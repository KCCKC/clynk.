ESP32 Arduino Firmware Code

Copy this sketch into the Arduino IDE and upload it to your ESP32. It automatically connects to your Wi-Fi and streams live sensor telemetry directly to this dashboard! 

#include <WiFi.h>
#include <PubSubClient.h>

// --- Wi-Fi Credentials ---
const char* ssid = "YOUR_WIFI_SSID";
const char* password = "YOUR_WIFI_PASSWORD";

// --- Free Public MQTT Broker ---
const char* mqtt_server = "broker.emqx.io";
const int mqtt_port = 1883;
const char* device_id = "device1"; // Must match your Clynk Device ID

WiFiClient espClient;
PubSubClient client(espClient);

void setup() {
  Serial.begin(115200);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected! IP: " + WiFi.localIP().toString());
  
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(callback);
}

void callback(char* topic, byte* payload, unsigned int length) {
  String message = "";
  for (int i = 0; i < length; i++) message += (char)payload[i];
  Serial.println("Received [" + String(topic) + "]: " + message);
  
  // Handle Relay 1 on Pin V2
  if (String(topic).endsWith("/V2")) {
    int val = message.toInt();
    digitalWrite(2, val ? HIGH : LOW); // Builtin LED / Relay
  }
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP32Client-" + String(random(0xffff), HEX);
    if (client.connect(clientId.c_str())) {
      Serial.println("Connected to Clynk Cloud!");
      String subTopic = "clynk/" + String(device_id) + "/#";
      client.subscribe(subTopic.c_str());
    } else {
      delay(2000);
    }
  }
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  static unsigned long lastMsg = 0;
  if (millis() - lastMsg > 1000) { // Stream every 1 second
    lastMsg = millis();
    
    // Simulate or read real DHT22/BME280 sensor data
    float temperature = 24.0 + (random(0, 30) / 10.0);
    float humidity = 55.0 + (random(0, 50) / 10.0);
    float voltage = 3.3 + (random(-10, 10) / 100.0);
    
    // Publish to Clynk Virtual Pins
    client.publish(("clynk/" + String(device_id) + "/V0").c_str(), String(temperature, 1).c_str());
    client.publish(("clynk/" + String(device_id) + "/V1").c_str(), String(humidity, 1).c_str());
    client.publish(("clynk/" + String(device_id) + "/V4").c_str(), String(voltage, 2).c_str());
    
    Serial.printf("Published: Temp=%.1f C, Hum=%.1f %%\n", temperature, humidity);
  }
}
